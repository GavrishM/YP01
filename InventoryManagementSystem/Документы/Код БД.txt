create database inventory_managment_system_BD_GavrishM;

create table categories (
id int primary key,
name text
);

create table suppliers (
id int primary key,
name text,
phone text,
email text
);

create table options_for_selling (
id int primary key,
name text
);

create table storage_methods (
id int primary key,
name text
);

create table products (
id int primary key,
name text,
product_range text,
category_id int,
supplier_id int,
option_for_selling_id int,
storage_method_id int,
foreign key (category_id)
references categories (id),
foreign key (supplier_id)
references suppliers (id),
foreign key (option_for_selling_id)
references options_for_selling (id),
foreign key (storage_method_id)
references storage_methods (id)
);

create table products_is_in_stock (
id int primary key,
product_id int,
quantity_in_stock numeric(10, 3),
quantity_for_sale numeric(10, 3),
foreign key (product_id)
references products (id)
);

create table users (
login text primary key,
password text,
name text,
email text,
phone text
);

create table log (
id int primary key,
product_id int,
type text,
quantity numeric(10, 3)
);
